function startVoice() { const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; if (!SpeechRecognition) { statusText.innerText = "Voice recognition is not supported in this browser. Please use Google Chrome."; return; } const recognition = new SpeechRecognition(); recognition.lang = languageSelect.value; recognition.continuous = false; recognition.interimResults = true; recognition.maxAlternatives = 1; let finalTranscript = ""; statusText.innerText = languageSelect.value === "te-IN" ? "🎤 తెలుగు వినబడుతోంది... మాట్లాడండి." : "🎤 Listening... Speak now."; recognition.onresult = function(event) { let interimTranscript = ""; for (let i = event.resultIndex; i < event.results.length; i++) { const transcript = event.results[i][0].transcript; if (event.results[i].isFinal) { finalTranscript += transcript; } else { interimTranscript += transcript; } } messageInput.value = (finalTranscript + " " + interimTranscript).trim(); }; recognition.onerror = function(event) { console.error("Voice recognition error:", event.error); if (event.error === "not-allowed") { statusText.innerText = "Please allow microphone access in Chrome."; } else if (event.error === "no-speech") { statusText.innerText = "No speech detected. Please try again."; } else { statusText.innerText = "Voice recognition failed. Please try again or type your question."; } }; recognition.onend = function() { messageInput.value = finalTranscript.trim() || messageInput.value; if (messageInput.value.trim()) { statusText.innerText = "Voice received. Sending your question..."; sendMessage(); } else if (!statusText.innerText.includes("Please allow")) { statusText.innerText = "No words recognized. Please try again or type your question."; } }; try { recognition.start(); } catch (error) { statusText.innerText = "Unable to start voice recognition. Please try again."; console.error(error); } }
